# MySQL -> SQLite translation cases.
#   @ <sql>   SQLite statement run on the scratch :memory: database
#   > <sql>   MySQL input
#   = <sql>   expected translation, must prepare on the scratch database ("=" alone: no-op)
#   ~ <sql>   expected translation, text only
#   -> <val>  first column of the first row of the translated input

# identifiers
> SELECT `key`,`value` FROM `t` WHERE `a` = ?
= SELECT "key","value" FROM "t" WHERE "a" = ?
> SELECT `we``ird`
~ SELECT "we`ird"
> SELECT t.key, t.value FROM t
= SELECT t."key", t.value FROM t

# CHAR_SEL_ENUM (reserved word after alias)
> SELECT c.guid, c.name, cp.entry FROM characters AS c LEFT JOIN character_pet AS cp ON c.guid = cp.owner AND cp.slot = ? WHERE c.account = ? AND c.deleteInfos_Name IS NULL ORDER BY COALESCE(c.order, c.guid)
= SELECT c.guid, c.name, cp.entry FROM characters AS c LEFT JOIN character_pet AS cp ON c.guid = cp.owner AND cp.slot = ? WHERE c.account = ? AND c.deleteInfos_Name IS NULL ORDER BY COALESCE(c."order", c.guid)

# ON DUPLICATE KEY UPDATE

> INSERT INTO guild_member_withdraw (guid, tab0, tab1, tab2, tab3, tab4, tab5, money) VALUES (?, ?, ?, ?, ?, ?, ?, ?) ON DUPLICATE KEY UPDATE tab0 = VALUES (tab0), tab1 = VALUES (tab1), tab2 = VALUES (tab2), tab3 = VALUES (tab3), tab4 = VALUES (tab4), tab5 = VALUES (tab5)
= INSERT INTO guild_member_withdraw (guid, tab0, tab1, tab2, tab3, tab4, tab5, money) VALUES (?, ?, ?, ?, ?, ?, ?, ?) ON CONFLICT DO UPDATE SET tab0 = excluded.tab0, tab1 = excluded.tab1, tab2 = excluded.tab2, tab3 = excluded.tab3, tab4 = excluded.tab4, tab5 = excluded.tab5
> INSERT INTO character_spell (guid, spell, specMask) VALUES (?, ?, ?) ON DUPLICATE KEY UPDATE specMask = VALUES(specMask)
= INSERT INTO character_spell (guid, spell, specMask) VALUES (?, ?, ?) ON CONFLICT DO UPDATE SET specMask = excluded.specMask
> INSERT INTO rbac_account_permissions (accountId, permissionId, granted, realmId) VALUES (?, ?, ?, ?) ON DUPLICATE KEY UPDATE granted = VALUES(granted)

> INSERT INTO `t` (`a`, `value`) VALUES (1, 'x') ON DUPLICATE KEY UPDATE `value` = VALUES(`value`), b = IF(b > 1, b + 1, 0)
= INSERT INTO "t" ("a", "value") VALUES (1, 'x') ON CONFLICT DO UPDATE SET "value" = excluded."value", b = IF(b > 1, b + 1, 0)
> INSERT INTO u (a, b) SELECT a, b FROM t ON DUPLICATE KEY UPDATE b = VALUES(b)
= INSERT INTO u (a, b) SELECT a, b FROM t WHERE true ON CONFLICT DO UPDATE SET b = excluded.b
> INSERT INTO u (a, b) SELECT a, COUNT(*) FROM t GROUP BY a ON DUPLICATE KEY UPDATE b = VALUES(b)
= INSERT INTO u (a, b) SELECT a, COUNT(*) FROM t WHERE true GROUP BY a ON CONFLICT DO UPDATE SET b = excluded.b
> INSERT INTO u (a, b) SELECT a, b FROM t WHERE a > 0 ON DUPLICATE KEY UPDATE b = u.b + 1
= INSERT INTO u (a, b) SELECT a, b FROM t WHERE a > 0 ON CONFLICT DO UPDATE SET b = u.b + 1

# DESC / SHOW
> DESC character_pet
= SELECT name FROM pragma_table_info('character_pet') ORDER BY cid
-> id
> DESCRIBE `characters`
= SELECT name FROM pragma_table_info('characters') ORDER BY cid
-> guid
> SHOW COLUMNS FROM t
= SELECT name FROM pragma_table_info('t') ORDER BY cid
> SHOW FULL FIELDS FROM t FROM main
= SELECT name FROM pragma_table_info('t', 'main') ORDER BY cid
> SHOW TABLES LIKE 'gm\_%'
= SELECT name FROM sqlite_schema WHERE type = 'table' AND name NOT LIKE 'sqlite\_%' ESCAPE '\' AND name LIKE 'gm\_%' ESCAPE '\' ORDER BY name
-> gm_ticket
> SHOW TABLES
= SELECT name FROM sqlite_schema WHERE type = 'table' AND name NOT LIKE 'sqlite\_%' ESCAPE '\' ORDER BY name
-> channels

# division, DIV, MOD
> SELECT a / b FROM t
= SELECT a * 1.0 / b FROM t
> SELECT 7/2
= SELECT 7 * 1.0 / 2
-> 3.5
> SELECT (a+1)/(b-1), SUM(b)/COUNT(*) FROM t

> SELECT 7 MOD 3

# modifiers and hints
> SELECT SQL_NO_CACHE a FROM t FORCE INDEX (PRIMARY) WHERE a = 1 FOR UPDATE
= SELECT a FROM t WHERE a = 1
> SELECT a FROM t USE INDEX FOR JOIN (x) LOCK IN SHARE MODE
= SELECT a FROM t
> SELECT STRAIGHT_JOIN t.a FROM t STRAIGHT_JOIN u ON t.a = u.a
= SELECT t.a FROM t JOIN u ON t.a = u.a
> SELECT 1 FROM DUAL
= SELECT 1

> SELECT ISNULL(b) FROM t WHERE a = 2
